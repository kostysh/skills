#!/usr/bin/env python3
"""Temporary command capture for the authorized framework trials; not a shipped runtime."""
import argparse,datetime,json,os,re,signal,subprocess,sys,time
from pathlib import Path
p=argparse.ArgumentParser();p.add_argument('--cwd',default='.');p.add_argument('--log',default='commands.jsonl');p.add_argument('--timeout',type=int,default=300);p.add_argument('command',nargs=argparse.REMAINDER);a=p.parse_args();cmd=a.command[1:] if a.command[:1]==['--'] else a.command
if not cmd:p.error('command required')
env=os.environ.copy();env.update({'SUPABASE_TELEMETRY_DISABLED':'1','DO_NOT_TRACK':'1','NEXT_TELEMETRY_DISABLED':'1','npm_config_cache':'/tmp/framework-platforms-20260909/cache/npm','XDG_CACHE_HOME':'/tmp/framework-platforms-20260909/cache','ELECTRON_CACHE':'/tmp/framework-platforms-20260909/cache/electron','PLAYWRIGHT_BROWSERS_PATH':'/tmp/framework-platforms-20260909/cache/playwright'})
def scrub(s):
 s=re.sub(r'eyJ[A-Za-z0-9_-]{8,}\.[A-Za-z0-9_-]+\.[A-Za-z0-9_-]+','[REDACTED_JWT]',s)
 s=re.sub(r'sb_secret_[A-Za-z0-9_-]+','[REDACTED_SECRET]',s)
 s=re.sub(r'(postgres(?:ql)?://[^:\s]+:)[^@\s]+@',r'\1[REDACTED]@',s)
 return s
start=time.monotonic();proc=subprocess.Popen(cmd,cwd=a.cwd,env=env,stdout=subprocess.PIPE,stderr=subprocess.PIPE,text=True,start_new_session=True)
try:out,err=proc.communicate(timeout=a.timeout);status=proc.returncode
except subprocess.TimeoutExpired:
 os.killpg(proc.pid,signal.SIGKILL);out,err=proc.communicate();status=124;err+='\nRecorder timeout; task process group killed.\n'
out,err=scrub(out),scrub(err)
event={'time':datetime.datetime.now(datetime.timezone.utc).isoformat(),'cwd':str(Path(a.cwd).resolve()),'command':[scrub(x) for x in cmd],'exit':status,'stdout':out,'stderr':err,'elapsed_seconds':round(time.monotonic()-start,3),'redaction':'JWT, secret API key and PostgreSQL URI password patterns'}
with Path(a.log).open('a') as f:f.write(json.dumps(event,ensure_ascii=False)+'\n')
sys.stdout.write(out);sys.stderr.write(err);print('[recorded exit='+str(status)+']',file=sys.stderr);sys.exit(status)
